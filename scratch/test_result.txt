            const engine = new DungeonEngine(state);
            results.push('DungeonEngine constructed successfully');

            results.push('Entered dungeon: ' + entered);

            // Simulate 10 frames
            for (let i = 0; i < 10; i++) {
                engine._update(0.016);
            }
            results.push('Simulated 10 update frames without errors');

            engine._render();

            engine._onKeyDown({ key: 'z' });

            results.push('Key Z toggled back to: ' + engine.targetZoom);
