        const logEl = document.getElementById('test-log');
        const results = [];

        function log(msg, status = 'INFO') {
            const time = new Date().toISOString().substring(11, 19);
            const line = `[${time}] [${status}] ${msg}`;
            console.log(line);
            logEl.textContent += line + '\n';
        }

        function assert(condition, testName, details = '') {
            if (condition) {
                log(`PASS: ${testName} ${details ? '(' + details + ')' : ''}`, 'SUCCESS');
                results.push({ test: testName, status: 'PASS', details });
            } else {
                log(`FAIL: ${testName} ${details ? '(' + details + ')' : ''}`, 'ERROR');
                results.push({ test: testName, status: 'FAIL', details });
                throw new Error(`Assertion failed: ${testName} - ${details}`);
            }
        }

                    'class_heretic', 'flayed_brute', 'ashen_monk', 'void_skitterer', 
                    'ironbound_executioner', 'combat_vfx'
                ];
                for (const key of testKeys) {
                    const st = AssetManager.status(key);
                    assert(st.loaded, `Asset [${key}] carregado`, `dim=${st.width}x${st.height}, type=${st.cachedType}`);
                    assert(st.width > 0 && st.height > 0, `Asset [${key}] dimens├Áes v├ílidas`);
                }

                // ÔöÇÔöÇ 2. STATE MANAGER & PERSIST├èNCIA CRUZADA ÔöÇÔöÇ
                log('Etapa 2: Inicializa├º├úo de Estado e Migra├º├úo de Save...');
                gameState.deleteSave();
                assert(!gameState.hasSave(), 'Save limpo com sucesso');

                gameState.initNewGame('warrior', 'pilgrim', 'flame_born');
                assert(gameState.character !== null, 'Campe├úo instanciado com sucesso');
                assert(gameState.character.className.length > 0, 'Classe atribu├¡da corretamente', gameState.character.className);
                assert(gameState.character.bloodlineId === 'children_of_pyre', 'Linhagem atribu├¡da corretamente', gameState.character.bloodlineId);

                gameState.save();
                assert(gameState.hasSave(), 'Save persistido no localStorage');
                assert(localStorage.getItem('in_search_of_hope_save') !== null, 'Chave in_search_of_hope_save gravada');
                assert(localStorage.getItem('ashen_bastion_save') !== null, 'Chave ashen_bastion_save sincronizada');

                // ÔöÇÔöÇ 3. SCREEN SHAKE CENTRALIZADO ÔöÇÔöÇ

                assert(typeof window.ScreenShakeManager === 'object', 'ScreenShakeManager global registrado');
                ScreenShakeManager.boss();
                assert(ScreenShakeManager.timer > 0, 'ScreenShakeManager ativado com preset boss');
                ScreenShakeManager.stop();
                assert(ScreenShakeManager.timer === 0, 'ScreenShakeManager parado com seguran├ºa');

                // ÔöÇÔöÇ 4. FORJA ANCESTRAL (BLACKSMITH UPGRADES) ÔöÇÔöÇ
                log('Etapa 4: Teste de T├¬mpera na Forja (+0 -> +1 -> +2 -> +3)...');

                assert((weapon.upgradeLevel || 0) === 0, 'Arma inicia no n├¡vel +0');

                // Add resources for upgrade

                gameState.resources[ResourceType.GOLD] = 500;

                const up1 = bs.upgradeItem(weapon.id);
                assert(up1.success && up1.newLevel === 1, 'T├¬mpera para +1 bem-sucedida', `stats = ${JSON.stringify(weapon.stats)}`);
                const up2 = bs.upgradeItem(weapon.id);
                assert(up2.success && up2.newLevel === 2, 'T├¬mpera para +2 bem-sucedida');
                const up3 = bs.upgradeItem(weapon.id);
                assert(up3.success && up3.newLevel === 3, 'T├¬mpera para +3 bem-sucedida (100% de taxa garantida)');

                const canvas = document.getElementById('test-canvas');
                gameState.dungeonKeys = 3;
                const dungeon = new DungeonEngine(gameState);

                assert(dungeon.enemies.length > 0, 'Inimigos gerados no calabou├ºo', `total = ${dungeon.enemies.length}`);

                const targetEnemy = dungeon.enemies.find(e => !e.isBoss);
                assert(targetEnemy !== undefined, 'Inimigo comum localizado');
                const initialEnemyHp = targetEnemy.hp;
                
                // Mover player para pr├│ximo do inimigo

                log('RESULTADO_FINAL: VERTICAL_SLICE_VALIDATED_SUCCESS', 'SUCCESS');
